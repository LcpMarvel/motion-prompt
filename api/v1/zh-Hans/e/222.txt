// 流程图 / Flowchart
// 包: diagram (资料中常用的图解与图表名称) · element 06 · 分类: 整理思路的图
// 画面内容: 开始→步骤→菱形判断→步骤→结束排成一行，判断处有一条回到首步的回环箭头；随后一个圆点沿路径行走，回环一次后走到结束。
// 运动规格:
//   - Shapes: start and end are 66×36 px pills (gray), steps are 72×40 px boxes (light blue), the decision is a diamond 92 px wide and 72 px tall (blue, white "?"), all in one row at y = 80
//   - Order: they pop in at 0.4, 0.8, 1.2, 1.6 and 1.95 s (0.3 s, ease-out-back); the arrows between them grow over 0.22 s from 0.6, 1.0, 1.4 and 1.75 s, with 10 px heads
//   - Loop: an arrow from the bottom of the diamond down to y = 172 and back up into the first step draws from 2.0 s to 2.45 s
//   - Marker: a dark gray dot (radius 8 px) follows start → step → decision → loop → step → decision → step → end from 2.6 s to 3.8 s (cubic ease-in-out), appearing 2.5–2.65 s and leaving 3.8–4.0 s
//   - End: the whole figure fades out from 4.1 s to 4.5 s and the card stays empty until the 4.7 s loop
// 适用场景: To show a procedure step by step, including where it branches on a condition.
// 标签: flowchart, flow chart, process, decision, diamond, steps, procedure, yes no
// 尺寸: 539x208 px · 循环: 4.7 s
// Needs: kit diagram -> https://lcpmarvel.github.io/motion-prompt/api/v1/kit/diagram.js
// 动态示例: https://lcpmarvel.github.io/motion-prompt/#222
//
// cells[i](k, w, h, u)……k = CanvasRenderingContext2D（调用方已清空画布并包好 save/restore），w、h = 尺寸，u = 秒（0 ≤ u < 循环时长）。画面仅由 u 决定。
// 以下为元素代码。CELLS、常量与辅助函数位于 kit 中。

// @element 06
// 06 フローチャート: start -> step -> decision (diamond) -> step -> end, with a "No" loop back;
// then a marker walks the path once (loops back once, then goes on to the end)
function flow(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const y = 80, S = [64, y], A = [158, y], D = [262, y], B = [364, y], E = [460, y], LY = 172;
  const dw = 46, dh = 36;                                   // diamond half sizes
  const arrows = [
    [[S[0] + 34, y], [A[0] - 36, y]],
    [[A[0] + 36, y], [D[0] - dw, y]],
    [[D[0] + dw, y], [B[0] - 36, y]],
    [[B[0] + 36, y], [E[0] - 34, y]],
  ];
  const loop = [[D[0], y + dh], [D[0], LY], [A[0], LY], [A[0], y + 20]];
  const tA = [0.6, 1.0, 1.4, 1.75];
  pen(k, C.g4, 3);
  arrows.forEach((pts, i) => trace(k, pts, easeInOut(P(u, tA[i], tA[i] + 0.22)), 10, C.g4));
  trace(k, loop, easeInOut(P(u, 2.0, 2.45)), 10, C.g4);
  // shapes
  const pill = (c, s, col) => box(k, c[0], c[1], 66, 36, 18, s, col);
  pill(S, pop(u, 0.4), C.g3);
  box(k, A[0], A[1], 72, 40, 6, pop(u, 0.8), C.blueL);
  const ds = pop(u, 1.2);
  if (ds > 0.01) {
    k.save(); k.translate(D[0], D[1]); k.scale(ds, ds); k.fillStyle = C.blue;
    k.beginPath(); k.moveTo(0, -dh); k.lineTo(dw, 0); k.lineTo(0, dh); k.lineTo(-dw, 0); k.closePath(); k.fill();
    txt(k, '?', 0, 2, `700 26px ${FONT}`, '#FFFFFF', 'center', 'middle');
    k.restore();
  }
  box(k, B[0], B[1], 72, 40, 6, pop(u, 1.6), C.blueL);
  pill(E, pop(u, 1.95), C.g3);
  // marker: start -> A -> D -> (No) loop -> A -> D -> (Yes) -> B -> E
  const path = [S, A, D, [D[0], LY], [A[0], LY], A, D, B, E];
  const m = P(u, 2.6, 3.8);
  const a = Math.min(P(u, 2.5, 2.65), 1 - P(u, 3.8, 4.0));
  if (a > 0) {
    const q = along(path, easeInOut(m) * pathLen(path));
    k.save(); k.globalAlpha *= a; k.fillStyle = C.mark; k.beginPath(); k.arc(q.x, q.y, 8, 0, TAU); k.fill(); k.restore();
  }
  k.globalAlpha = 1;
}
CELLS[5] = flow;
